1.两个线程交替打印
    private static List<Integer> intList = new ArrayList<>(Arrays.asList(1, 2, 3, 4, 5, 6));
    private static List<String> strList = new ArrayList<>(Arrays.asList("A", "B", "C", "D", "E", "F"));

    private static ReentrantLock lock = new ReentrantLock();

    private static Condition condition = lock.newCondition();

    // true打印数字 false字符串
    private volatile static boolean printInt = true;

    public static void main(String[] args){

        new Thread(() -> {
            intList.forEach(f -> {
                print(String.valueOf(f), true);
            });
        }).start();

        new Thread(() -> {
            strList.forEach(f -> {
                print(f, false);
            });
        }).start();

    }

    public static void print(String str, boolean flag){
        lock.lock();
        try{
            // 检查当前是否轮到自己执行
            while (printInt != flag) {
                condition.await(); // 不满足条件则等待
            }
            System.out.println(str); // 打印内容
            printInt = !printInt;     // 翻转标志位
            condition.signal();
        }catch (InterruptedException ex){
            // ignore
        }finally {
            lock.unlock();;
        }
    }
